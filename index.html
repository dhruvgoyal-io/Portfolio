# R Language // Quantitative Volatility & Log-Return Tail Analysis
library(stats)
set.seed(26007)
n_days <- 252
mu <- 0.14; sigma <- 0.24
returns <- rnorm(n_days, mean = mu/n_days, sd = sigma/sqrt(n_days))

sharpe_ratio <- (mean(returns) * 252) / (sd(returns) * sqrt(252))
var_95 <- quantile(returns, probs = 0.05)
cat(sprintf("Annualized Sharpe: %.3f\\n95%% Daily VaR: %.4f\\n", sharpe_ratio, var_95))
